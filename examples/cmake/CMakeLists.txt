cmake_minimum_required(VERSION 3.25)
project(sintergpu_consumer LANGUAGES CXX)
find_package(sintergpu 0.2 CONFIG REQUIRED)
find_package(Vulkan REQUIRED)
add_library(consumer OBJECT main.cpp)
target_link_libraries(consumer PRIVATE sintergpu::kernels Vulkan::Vulkan)
if(NOT EXISTS "${SINTERGPU_KERNEL_DIR}/common/block_reduce.slang")
  message(FATAL_ERROR "Installed shader resources missing")
endif()
if(NOT EXISTS "${SINTERGPU_DATA_DIR}/timings.json")
  message(FATAL_ERROR "Installed timing database missing")
endif()
file(READ "${SINTERGPU_DATA_DIR}/timings.json" _sintergpu_timings)
file(READ "${SINTERGPU_DATA_DIR}/catalog.json" _sintergpu_catalog)
string(JSON _sintergpu_timing_count LENGTH "${_sintergpu_timings}" operations)
string(JSON _sintergpu_catalog_count LENGTH "${_sintergpu_catalog}" operations)
if(NOT _sintergpu_timing_count EQUAL _sintergpu_catalog_count)
  message(FATAL_ERROR "Installed timing database does not cover the catalog")
endif()
